cmake_minimum_required(VERSION 3.25)
project(Resonant VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
include(CTest)
include(GNUInstallDirs)

option(RESONANT_SANITIZE "Enable ASan and UBSan" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -ffp-contract=off)
  if(RESONANT_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()

add_library(resonant_core src/wave.cpp src/bytecode.cpp src/parser.cpp src/compiler.cpp src/dsp.cpp src/simd.cpp src/vm.cpp src/engine.cpp src/host_clock.cpp src/midi.cpp src/session.cpp src/files.cpp src/spectral.cpp src/effects.cpp)
target_include_directories(resonant_core PUBLIC include)

function(resonant_test name)
  if(BUILD_TESTING)
    add_executable(test_${name} tests/${name}.cpp)
    target_link_libraries(test_${name} PRIVATE resonant_core)
    add_test(NAME ${name} COMMAND test_${name})
  endif()
endfunction()

resonant_test(wave)

resonant_test(bytecode)
resonant_test(parser)
resonant_test(compiler)
resonant_test(vm)
find_package(Threads REQUIRED)
target_link_libraries(resonant_core PUBLIC Threads::Threads)
resonant_test(engine)
if(BUILD_TESTING)
  set_tests_properties(engine PROPERTIES TIMEOUT 30)
endif()
resonant_test(session)
resonant_test(host_clock)

option(RESONANT_DEVICE "Build the PortAudio device adapter when available" ON)
if(RESONANT_DEVICE)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(PORTAUDIO QUIET IMPORTED_TARGET portaudio-2.0>=19)
  endif()
endif()
target_sources(resonant_core PRIVATE src/device.cpp)
if(RESONANT_DEVICE AND TARGET PkgConfig::PORTAUDIO)
  target_compile_definitions(resonant_core PRIVATE RESONANT_PORTAUDIO=1)
  target_link_libraries(resonant_core PRIVATE PkgConfig::PORTAUDIO)
else()
  target_compile_definitions(resonant_core PRIVATE RESONANT_PORTAUDIO=0)
endif()

add_executable(resonant src/cli.cpp)
target_link_libraries(resonant PRIVATE resonant_core)
install(TARGETS resonant RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

resonant_test(spectral)
resonant_test(effects)

option(RESONANT_WORKBENCH "Build the native AppKit workbench on macOS" ON)
if(APPLE AND RESONANT_WORKBENCH)
  enable_language(OBJCXX)
  set(starter compositions/ember-motif.rsn)
  set_source_files_properties(${starter} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  add_library(resonant_core_midi STATIC src/apple/midi_input.mm)
  set_target_properties(resonant_core_midi PROPERTIES
    OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS NO)
  target_compile_options(resonant_core_midi PRIVATE -fobjc-arc)
  target_link_libraries(resonant_core_midi PUBLIC resonant_core "-framework CoreMIDI" "-framework CoreFoundation")
  add_executable(resonant_workbench MACOSX_BUNDLE src/apple/workbench.mm src/apple/views.mm ${starter})
  set_target_properties(resonant_workbench PROPERTIES
    OUTPUT_NAME Resonant
    OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS NO
    MACOSX_BUNDLE_GUI_IDENTIFIER org.resonant.workbench
    MACOSX_BUNDLE_BUNDLE_NAME Resonant
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
  target_compile_options(resonant_workbench PRIVATE -fobjc-arc)
  target_link_libraries(resonant_workbench PRIVATE resonant_core_midi "-framework AppKit")
  install(TARGETS resonant_workbench BUNDLE DESTINATION .)
  if(BUILD_TESTING)
    add_executable(test_core_midi tests/core_midi.mm)
    set_target_properties(test_core_midi PROPERTIES
      OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS NO)
    target_compile_options(test_core_midi PRIVATE -fobjc-arc)
    target_link_libraries(test_core_midi PRIVATE resonant_core_midi)
  endif()
endif()

option(RESONANT_FUZZ "Build the Clang libFuzzer diagnostic target" OFF)
if(RESONANT_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The fuzz target requires a Clang toolchain with libFuzzer")
  endif()
  target_compile_options(resonant_core PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(resonant_fuzz tests/fuzz.cpp)
  target_compile_options(resonant_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(resonant_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_libraries(resonant_fuzz PRIVATE resonant_core)
endif()
